{\centering\section*{Введение}}\addcontentsline{toc}{section}{Введение}

Цель работы -- получение практических навыков работы с мультимножествами, изучение алгоритма генерации бинарного кода Грея, а также реализация данных структур данных и алгоритмов в виде модульной программы на языке программирования C++.

Задание: разработать программу генерации бинарного кода Грея для формирования элементов универсума мультимножеств заданной пользователем разрядности $n \in [0, 20]$. На основе созданного универсума формируются два мультимножества $A$ и $B$ (вручную или автоматически по выбору пользователя) с заданными мощностями. Программа должна вычислять теоретико-множественные операции (объединение, пересечение, обе разности, симметрическую разность, дополнения до универсума) и арифметические операции (сумму, разности, произведение и деление). Также необходимо реализовать защиту от некорректного пользовательского ввода и интерактивное консольное меню.

{\centering\section{Математическое описание}}

{\subsection{Множества и мультимножества}}
Множество -- это любая определенная совокупность объектов, называемых его элементами, причем все элементы множества по определению попарно различимы. Множество, не содержащее ни одного элемента, называется пустым и обозначается как $\varnothing$.

Мультимножеством называется математический объект, в котором допускается наличие повторяющихся (одинаковых) элементов. Пусть дано базовое конечное множество носителя $X = \{x_1, \dots, x_n\}$. Тогда мультимножеством $\widehat{X}$ над множеством $X$ называется совокупность:
\[
    \widehat{X} = [x_1^{a_1}, \dots, x_n^{a_n}] = \langle a_1(x_1), \dots, a_n(x_n) \rangle, \quad a_i \geqslant 0,
\]
где $a_i = k_{\widehat{X}}(x_i)$ -- кратность вхождения элемента $x_i$. Множество $X$ называют \textit{носителем}, подмножество $\underline{\widehat{X}} = \{x_i \in X \mid a_i > 0\}$ -- \textit{составом}, а величину $m = \sum_{i=1}^n a_i$ -- \textit{мощностью} мультимножества.

Универсумом $U$ в рамках работы называется мультимножество над алфавитом кодов Грея, содержащее все возможные кодовые комбинации заданной разрядности $n$ со случайными максимальными кратностями $k_U(x) \in [1, 7]$. Мультимножество $A$ является корректным подмультимножеством ($A \subseteq U$), если $\forall x \in X: k_A(x) \leqslant k_U(x)$.

{\subsection{Алгоритм генерации бинарного кода Грея}}
$n$-разрядный код Грея представляет собой перечисление всех $2^n$ двоичных векторов длины $n$, упорядоченных так, чтобы любые два соседних значения различались ровно в одном бите (расстояние Хэмминга равно 1). Для прямого нерекурсивного построения $i$-го элемента используется побитовая операция:
\begin{equation}
    G(i) = i \oplus (i \gg 1), \quad i = 0, \dots, 2^n - 1.
\end{equation}

{\subsection{Структуры данных}}
Для инкапсуляции мультимножеств спроектирован класс \texttt{MultiSet}. Внутренним хранилищем выступает ассоциативный контейнер \texttt{std::map<std::string, std::size\_t>}. Это гарантирует автоматическую сортировку элементов по строковому представлению кодов Грея и исключает хранение элементов с нулевой кратностью.

\begin{lstlisting}[caption={Класс мультимножества \texttt{MultiSet}}, label={lst:multiset}]
class MultiSet
{
public:
    using Count = std::size_t;
    using Data = std::map<std::string, Count>;
    MultiSet() = default;
    void clear();
    void add(const std::string &value, Count count = 1);
    Count count(const std::string &value) const;
    Count size() const;
    bool empty() const;
    const Data &data() const;
private:
    Data data_;
};
\end{lstlisting}

{\subsection{Описание функций}}

{\subsubsection{decimalToBinary и generateGrayCode}}
Функция \texttt{decimalToBinary} переводит беззнаковое целое число в двоичную строку заданной фиксированной разрядности с сохранением ведущих нулей. Функция \texttt{generateGrayCode} последовательно формирует вектор всех $2^n$ кодов. Для исключения переполнения знакового типа при сдвигах используется беззнаковое приведение \texttt{static\_cast<std::size\_t>(1) << bits}.

\begin{lstlisting}[caption={Генерация кодов Грея}, label={lst:gray}]
std::string decimalToBinary(unsigned long long value, std::size_t width) {
    std::string result(width, '0');
    for (std::size_t i = 0; i < width; ++i) {
        const std::size_t shift = width - 1 - i;
        result[i] = ((value >> shift) & 1ULL) ? '1' : '0';
    }
    return result;
}

std::vector<std::string> generateGrayCode(std::size_t bits) {
    std::vector<std::string> result;
    if (bits == 0) return result;
    const std::size_t count = static_cast<std::size_t>(1) << bits;
    result.reserve(count);
    for (std::size_t i = 0; i < count; ++i) {
        const unsigned long long gray = i ^ (i >> 1);
        result.push_back(decimalToBinary(gray, bits));
    }
    return result;
}
\end{lstlisting}

{\subsubsection{Реализация операций над мультимножествами}}
Все операции реализованы в функциональном стиле: принимают константные ссылки на операнды и возвращают новый экземпляр \texttt{MultiSet}.

\begin{lstlisting}[caption={Операции над мультимножествами}, label={lst:ops}]
MultiSet setUnion(const MultiSet &a, const MultiSet &b) {
    MultiSet result = a;
    for (const auto &[value, countB] : b.data()) {
        const auto countA = result.count(value);
        if (countB > countA) result.add(value, countB - countA);
    }
    return result;
}

MultiSet intersection(const MultiSet &a, const MultiSet &b) {
    MultiSet result;
    for (const auto &[value, countA] : a.data()) {
        const auto countB = b.count(value);
        const auto count = std::min(countA, countB);
        if (count > 0) result.add(value, count);
    }
    return result;
}

MultiSet complement(const MultiSet &a, const MultiSet &universe) {
    MultiSet result;
    for (const auto &[value, countU] : universe.data()) {
        const auto countA = a.count(value);
        if (countU > countA) result.add(value, countU - countA);
    }
    return result;
}

MultiSet difference(const MultiSet &a, const MultiSet &b, const MultiSet &universe) {
    return intersection(a, complement(b, universe));
}

MultiSet symmetricDifference(const MultiSet &a, const MultiSet &b, const MultiSet &universe) {
    return difference(setUnion(a, b), intersection(a, b), universe);
}
\end{lstlisting}

{\subsection{Алгоритм автоматического заполнения}}
Автоматическое заполнение мультимножества мощности $p$ выполняется методом случайной выборки без возвращения из общего пула элементов универсума. Вектор \texttt{pool} предварительно заполняется ссылками на элементы (каждый элемент дублируется столько раз, какова его кратность в $U$), перемешивается генератором \texttt{std::mt19937\_64} (\texttt{std::shuffle}) и первые $p$ значений вставляются в результирующее мультимножество. Сложность алгоритма составляет $O(|U| + p \log |X|)$.

{\subsection{Ручное заполнение мультимножеств}}
При ручном заполнении пользователь выбирает элементы по их порядковым номерам из выведенного списка кодов Грея. При каждом выборе программа проверяет:
\begin{enumerate}
    \item принадлежность индекса допустимому диапазону $[1, |X|]$;
    \item текущую накопленную кратность выбранного кода: если она уже сравнялась с кратностью данного элемента в универсуме $U$, программа отклоняет ввод и сообщает об исчерпании лимита.
\end{enumerate}

{\centering\section*{Заключение}}\addcontentsline{toc}{section}{Заключение}

В ходе выполнения лабораторной работы были изучены мультимножества, операции над ними и свойства бинарных кодов Грея. Разработанный программный комплекс на языке C++ обеспечивает:
\begin{itemize}
    \item генерацию бинарного кода Грея разрядности $n \in [0, 20]$;
    \item построение универсума со случайными кратностями $k_U(x) \in [1, 7]$;
    \item корректное ручное и автоматическое заполнение подмультимножеств;
    \item расчет теоретико-множественных и арифметических операций;
    \item строгий контроль типов и значений входных данных.
\end{itemize}

\textbf{Достоинства реализации:}
\begin{itemize}
    \item Инкапсуляция логики внутри класса \texttt{MultiSet};
    \item Автоматическое исключение нулевых кратностей за счет свойств \texttt{std::map};
    \item Независимость математических функций от способа консольного ввода.
\end{itemize}

\textbf{Недостатки и пути оптимизации:}
\begin{itemize}
    \item Логарифмический поиск в \texttt{std::map} при больших $n$ уступает по скорости плоскому индексированному вектору кратностей ($O(1)$ на доступ);
    \item Высокие накладные расходы по памяти при генерации пула строк в автоматическом заполнении при $n=20$.
\end{itemize}

#include "operations.hpp"
#include <algorithm>

MultiSet setUnion(const MultiSet &a, const MultiSet &b) {
    MultiSet result = a;
    for (const auto &[value, countB] : b.data()) {
        const auto countA = result.count(value);
        if (countB > countA) result.add(value, countB - countA);
    }
    return result;
}

MultiSet intersection(const MultiSet &a, const MultiSet &b) {
    MultiSet result;
    for (const auto &[value, countA] : a.data()) {
        const auto countB = b.count(value);
        const auto count = std::min(countA, countB);
        if (count > 0) result.add(value, count);
    }
    return result;
}

MultiSet complement(const MultiSet &a, const MultiSet &universe) {
    MultiSet result;
    for (const auto &[value, countU] : universe.data()) {
        const auto countA = a.count(value);
        if (countU > countA) result.add(value, countU - countA);
    }
    return result;
}

MultiSet difference(const MultiSet &a, const MultiSet &b, const MultiSet &universe) {
    return intersection(a, complement(b, universe));
}

MultiSet symmetricDifference(const MultiSet &a, const MultiSet &b, const MultiSet &universe) {
    return difference(setUnion(a, b), intersection(a, b), universe);
}

MultiSet arithmeticSum(const MultiSet &a, const MultiSet &b, const MultiSet &universe) {
    MultiSet result = a;
    for (const auto &[value, countB] : b.data()) {
        const auto current = result.count(value);
        const auto maxAllowed = universe.count(value);
        if (current < maxAllowed) {
            const auto target = std::min(current + countB, maxAllowed);
            result.add(value, target - current);
        }
    }
    return result;
}

MultiSet arithmeticDifference(const MultiSet &a, const MultiSet &b) {
    MultiSet result;
    for (const auto &[value, countA] : a.data()) {
        const auto countB = b.count(value);
        if (countA > countB) result.add(value, countA - countB);
    }
    return result;
}

MultiSet arithmeticProduct(const MultiSet &a, const MultiSet &b, const MultiSet &universe) {
    MultiSet result;
    for (const auto &[value, countA] : a.data()) {
        const auto countB = b.count(value);
        if (countB > 0) {
            const auto count = std::min(countA * countB, universe.count(value));
            result.add(value, count);
        }
    }
    return result;
}

MultiSet arithmeticDivision(const MultiSet &a, const MultiSet &b) {
    MultiSet result;
    for (const auto &[value, countA] : a.data()) {
        const auto countB = b.count(value);
        if (countB > 0 && countA >= countB) result.add(value, countA / countB);
    }
    return result;
}
